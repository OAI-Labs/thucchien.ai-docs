import ApiReferenceLayout, { LeftColumn, RightColumn } from '@site/src/components/ApiReferenceLayout';
import InteractiveApiRequest from '@site/src/components/InteractiveApiRequest';
import ApiParameter from '@site/src/components/ApiParameter';
import Badge from '@site/src/components/Badge';
import styles from '@site/src/components/InteractiveApiRequest.module.css';
import { useState, useEffect } from 'react';

export const KeyInfoPreview = ({ data, apiKey }) => {
  const info = data?.info;
  const teamId = info?.team_id;
  const [team, setTeam] = useState(null);
  const [models, setModels] = useState(null);

  // Budget, RPM, TPM được đặt ở team (dùng chung cho mọi key của đội), nên lấy thêm /team/info.
  useEffect(() => {
    setTeam(null);
    if (!teamId || !apiKey) return;
    fetch(`https://api.thucchien.ai/team/info?team_id=${encodeURIComponent(teamId)}`, {
      headers: { accept: 'application/json', Authorization: `Bearer ${apiKey}` },
    })
      .then((res) => (res.ok ? res.json() : null))
      .then((json) => setTeam(json?.team_info || null))
      .catch(() => setTeam(null));
  }, [teamId, apiKey]);

  // Key của đội dùng nhóm model "contestant-models", nên info.models rỗng. /v1/models trả đúng danh sách key gọi được.
  useEffect(() => {
    setModels(null);
    if (!info || !apiKey) return;
    fetch('https://api.thucchien.ai/v1/models', {
      headers: { accept: 'application/json', Authorization: `Bearer ${apiKey}` },
    })
      .then((res) => (res.ok ? res.json() : null))
      .then((json) => setModels(json?.data?.map((m) => m.id).sort() || null))
      .catch(() => setModels(null));
  }, [info, apiKey]);

  if (!info) return null;

  const spend = team?.spend ?? info.spend ?? 0;
  const maxBudget = team?.max_budget ?? info.max_budget;
  const tpmLimit = team?.tpm_limit ?? info.tpm_limit;
  const rpmLimit = team?.rpm_limit ?? info.rpm_limit;
  const modelTpm = team?.metadata?.model_tpm_limit || {};
  const modelRpm = team?.metadata?.model_rpm_limit || {};
  const limitedFirst = (models || []).slice().sort(
    (a, b) => (modelTpm[b] || modelRpm[b] ? 1 : 0) - (modelTpm[a] || modelRpm[a] ? 1 : 0)
  );
  const budgetPercentage = maxBudget ? Math.min((spend / maxBudget) * 100, 100) : 0;

  const getProgressBarColor = () => {
    if (!maxBudget) return styles.progressBarSuccess;
    if (budgetPercentage >= 90) return styles.progressBarDanger;
    if (budgetPercentage >= 70) return styles.progressBarWarning;
    return styles.progressBarSuccess;
  };

  return (
    <div className={styles.keyInfoPreview}>
      <div className={styles.keyInfoTitle}>
        📊 Budget Usage{team ? ` (team ${team.team_id})` : ''}
      </div>
      
      {/* Budget Progress Section */}
      <div className={styles.budgetSection}>
        <div className={styles.budgetHeader}>
          <div className={styles.budgetAmount}>
            <span className={styles.budgetLabel}>Spend:</span>
            <span className={styles.budgetValue}>${spend.toFixed(4)}</span>
          </div>
          <div className={styles.budgetAmount}>
            <span className={styles.budgetLabel}>Budget:</span>
            <span className={styles.budgetValue}>
              {maxBudget ? `$${maxBudget.toFixed(2)}` : 'Unlimited'}
            </span>
          </div>
        </div>
        
        {maxBudget ? (
          <>
            <div className={styles.progressBarContainer}>
              <div 
                className={`${styles.progressBar} ${getProgressBarColor()}`}
                style={{ width: `${budgetPercentage}%` }}
              >
                <span className={styles.progressBarText}>
                  {budgetPercentage.toFixed(1)}%
                </span>
              </div>
            </div>
          </>
        ) : (
          <div className={styles.unlimitedBadge}>
            No budget limit set
          </div>
        )}
      </div>

      {/* Additional Info Section */}
      <div className={styles.infoSection}>
        <div className={styles.infoRow}>
          <span className={styles.infoLabel}>Key:</span>
          <span className={styles.infoValue}>{info.key_alias || info.key_name || 'N/A'}</span>
        </div>
        <div className={styles.infoRow}>
          <span className={styles.infoLabel}>Status:</span>
          <span className={styles.infoValue}>
            {info.blocked ? (
              <span className={styles.statusBadgeDanger}>● Blocked</span>
            ) : (
              <span className={styles.statusBadgeSuccess}>● Active</span>
            )}
          </span>
        </div>
        <div className={styles.infoRow}>
          <span className={styles.infoLabel}>TPM Limit:</span>
          <span className={styles.infoValue}>
            {tpmLimit ? tpmLimit.toLocaleString() : 'Unlimited'}
          </span>
        </div>
        <div className={styles.infoRow}>
          <span className={styles.infoLabel}>RPM Limit:</span>
          <span className={styles.infoValue}>
            {rpmLimit ? rpmLimit.toLocaleString() : 'Unlimited'}
          </span>
        </div>
        <div className={styles.infoRow}>
          <span className={styles.infoLabel}>Models:</span>
          <span className={styles.infoValue}>
            {models ? `${models.length} model` : info.models?.join(', ') || 'N/A'}
          </span>
        </div>
      </div>

      {models && (
        <details className={styles.modelLimits} open>
          <summary>Giới hạn từng model</summary>
          <table>
            <thead>
              <tr><th>Model</th><th>TPM</th><th>RPM</th></tr>
            </thead>
            <tbody>
              {limitedFirst.map((m) => (
                <tr key={m}>
                  <td><code>{m}</code></td>
                  <td>{modelTpm[m] ? modelTpm[m].toLocaleString() : '—'}</td>
                  <td>{modelRpm[m] ? modelRpm[m].toLocaleString() : '—'}</td>
                </tr>
              ))}
            </tbody>
          </table>
          <div className={styles.modelLimitsNote}>
            — : chỉ chịu giới hạn chung của đội. Xem <a href="/docs/round-2/user-guide/rate-limits">Giới hạn tốc độ</a>.
          </div>
        </details>
      )}
    </div>
  );
};

# Kiểm tra chi tiêu (Spend Checking)

## <Badge method="GET" /> `/key/info`

Lấy thông tin chi tiết về API key và mức độ sử dụng hiện tại.

:::info Budget tính theo đội
Budget, RPM, TPM và giới hạn từng model được đặt cho **đội** (team), dùng chung cho mọi key của đội. Vì vậy trong `/key/info`, `max_budget`, `rpm_limit`, `tpm_limit` là `null`, `metadata` không có giới hạn từng model, và `spend` chỉ là phần chi tiêu của riêng key đó.

Để xem tổng chi tiêu và budget của đội, gọi `/team/info` với `team_id` lấy từ `/key/info` (chỉ xem được đội của chính mình). Khung kết quả bên phải đã tự gọi thêm endpoint này.

```bash
curl 'https://api.thucchien.ai/team/info?team_id=<team_id>'   -H 'Authorization: Bearer <your_api_key>'
```

Trong phản hồi, xem `team_info.spend`, `team_info.max_budget`, `team_info.rpm_limit`, `team_info.tpm_limit`, và giới hạn từng model ở `team_info.metadata.model_tpm_limit` / `team_info.metadata.model_rpm_limit`. Danh sách model key gọi được: `GET /v1/models`. Giải thích các giới hạn: [Giới hạn tốc độ](../user-guide/rate-limits.mdx).
:::

    <ApiParameter name="key_alias" type="string" indent={2}>

      Ngân sách tối đa riêng của API key này. `null` nếu key không có giới hạn riêng (budget đặt ở đội, xem `/team/info`).
    </ApiParameter>
        <ApiParameter name="max_parallel_requests" type="number" indent={2}>
      Số lượng yêu cầu đồng thời tối đa được phép cho API key này. Giới hạn này kiểm soát số lượng request có thể xử lý song song cùng một lúc. Trường hợp vượt quá giới hạn này, các yêu cầu bổ sung sẽ bị từ chối trả lỗi 429 (Too Many Requests).
    </ApiParameter>
    <ApiParameter name="metadata" type="object" indent={2}>
      Metadata tùy chỉnh được gán cho API key.
    </ApiParameter>
    <ApiParameter name="metadata.model_rpm_limit" type="object" indent={3}>
      Giới hạn request per minute (RPM) cho từng model. Với key của đội, giới hạn này nằm ở `team_info.metadata.model_rpm_limit` của `/team/info`, không nằm trên key.
    </ApiParameter>
    <ApiParameter name="metadata.model_tpm_limit" type="object" indent={3}>
      Giới hạn token per minute (TPM, input + output) cho từng model. Với key của đội, giới hạn này nằm ở `team_info.metadata.model_tpm_limit` của `/team/info`, không nằm trên key.
    </ApiParameter>
    <ApiParameter name="tpm_limit" type="number" indent={2}>
      Giới hạn tokens per minute chung cho API Key. `null` nếu không giới hạn.
    </ApiParameter>
    <ApiParameter name="rpm_limit" type="number" indent={2}>
      Giới hạn requests per minute chung cho API Key. `null` nếu không giới hạn.
    </ApiParameter>
    <ApiParameter name="expires" type="string" indent={2}>
      Thời gian hết hạn của API key. `null` nếu không hết hạn.
    </ApiParameter>
    <ApiParameter name="models" type="array" indent={2}>
